using System.Windows;

using Microsoft.EntityFrameworkCore;
namespace CoffeeRuu
{
    public partial class MainWindow : Window
    {
        private OrderService? _orderService;

        // khởi tạo order service
        private OrderService GetOrderModuleService()
        {
            if (_orderService != null)
            {
                return _orderService;
            }

            // Lấy connection string riêng của máy
            string? connectionString =
                Environment.GetEnvironmentVariable(
                    "COFFEERUU_CONNECTION");

            // Nếu không có thì lấy từ App.config
            if (string.IsNullOrWhiteSpace(connectionString))
            {
                connectionString =
                    ConfigurationManager.ConnectionStrings[
                        "CoffeeRuuConnection"]?.ConnectionString;
            }

            if (string.IsNullOrWhiteSpace(connectionString))
            {
                throw new InvalidOperationException(
                    "Không tìm thấy CoffeeRuuConnection.");
            }

            // Cấu hình Entity Framework Core
            var options =
                new DbContextOptionsBuilder<CoffeeRuuDbContext>()
                    .UseSqlServer(connectionString)
                    .Options;

            // Khởi tạo Repository
            var repository = new OrderRepository(options);

            // Khởi tạo Service
            _orderService = new OrderService(repository);

            return _orderService;
        }
        public MainWindow()
        {
            InitializeComponent();

            // 1. Tự động load trang chủ ngay khi mở cửa sổ chính
            MainContent.Navigate(new HomeView());
            MainContent.Visibility = Visibility.Visible;
        }

        private void BtnLogout_Click(object sender, RoutedEventArgs e)
        {
            MessageBoxResult result = MessageBox.Show(
                "Bạn có chắc chắn muốn đăng xuất?",
                "Xác nhận",
                MessageBoxButton.YesNo,
                MessageBoxImage.Question);

            if (result == MessageBoxResult.Yes)
            {
                LoginView login = new LoginView();
                Application.Current.MainWindow = login;
                login.Show();
                this.Close();
            }
        }

        // Chuyển về trang chủ
        private void BtnHome_Click(object sender, RoutedEventArgs e)
        {
            if (!(MainContent.Content is HomeView))
            {
                MainContent.Navigate(new HomeView());
            }
        }

        // Chuyển sang quản lý kho
        private void BtnInventory_Click(object sender, RoutedEventArgs e)
        {
            if (!(MainContent.Content is InventoryView))
            {
                MainContent.Navigate(new InventoryView());
            }
        }

        // Chuyển sang quản lý Bàn
        private void BtnTable_Click(object sender, RoutedEventArgs e)
        {
            if (!(MainContent.Content is TableView))
            {
                MainContent.Navigate(new TableView());
            }
        }

        // Chuyển sang giao diện Menu gọi món (bán hàng)
        private void BtnMenu_Click(object sender, RoutedEventArgs e)
        {
            if (!(MainContent.Content is PosMenuView))
            {
                MainContent.Navigate(new PosMenuView());
            }
        }



        // Chuyển sang quản lý sản phẩm
        private void BtnProduct_Click(object sender, RoutedEventArgs e)
        {
            if (!(MainContent.Content is ProductManagementView))
            {
                MainContent.Navigate(new ProductManagementView());
            }

            MainContent.Visibility = Visibility.Collapsed;
            MainContent.Visibility = Visibility.Visible;
        }


        // chuyển sang order processing

        private void OpenOrders_Click(
            object sender,
            RoutedEventArgs e)
        {
            if (MainContent.Content is OrderView)
            {
                return;
            }

            var orderViewModel = new OrderViewModel(
                GetOrderModuleService());

            var orderPage = new OrderView
            {
                DataContext = orderViewModel
            };

            MainContent.Navigate(orderPage);
        }


        // Chuyển sang màn hình báo cáo (Analytics & Reports)
        private void BtnReport_Click(object sender, RoutedEventArgs e)
        {
            if (!(MainContent.Content is ReportView))
            {
                MainContent.Navigate(new ReportView());
            }
        }


        // chuyển sang kds pha chế

        private void OpenKds_Click(
            object sender,
            RoutedEventArgs e)
        {
            if (MainContent.Content is KdsView)
            {
                return;
            }

            var kdsViewModel = new KdsViewModel(
                GetOrderModuleService());

            var kdsPage = new KdsView
            {
                DataContext = kdsViewModel
            };

            MainContent.Navigate(kdsPage);
        }

    }

        {
            _contextFactory = contextFactory;
        }

        // 1. Lấy tất cả nguyên liệu
        public async Task<List<Ingredient>> GetAllAsync()
        {
            using var context = _contextFactory();

            return await context.Ingredients
                .AsNoTracking()
                .OrderBy(i => i.IngredientName)
                .ToListAsync();
        }

        // 2. Tìm nguyên liệu theo ID
        public async Task<Ingredient?> GetByIdAsync(int id)
        {
            using var context = _contextFactory();

            return await context.Ingredients
                .AsNoTracking()
                .FirstOrDefaultAsync(i => i.IngredientId == id);
        }

        // 3. Kiểm tra tên nguyên liệu đã tồn tại
        public async Task<bool> NameExistsAsync(
            string name, int? excludeId = null)
        {
            using var context = _contextFactory();

            return await context.Ingredients
                .AnyAsync(i =>
                    i.IngredientName == name &&
                    (!excludeId.HasValue ||
                     i.IngredientId != excludeId.Value));
        }

        // 4. Thêm nguyên liệu
        public async Task AddAsync(Ingredient ingredient)
        {
            using var context = _contextFactory();

            context.Ingredients.Add(ingredient);

            await context.SaveChangesAsync();
        }

        // 5. Cập nhật thông tin nguyên liệu
        public async Task<bool> UpdateInfoAsync(
            int id,
            string name,
            string unit,
            decimal minimumStock)
        {
            using var context = _contextFactory();

            var ingredient = await context.Ingredients
                .FirstOrDefaultAsync(i => i.IngredientId == id);

            if (ingredient == null)
                return false;

            ingredient.IngredientName = name;
            ingredient.Unit = unit;
            ingredient.MinimumStock = minimumStock;

            await context.SaveChangesAsync();

            return true;
        }

        // 6. Thay đổi trạng thái hoạt động
        public async Task<bool> SetActiveAsync(
            int id, bool isActive)
        {
            using var context = _contextFactory();

            var ingredient = await context.Ingredients
                .FirstOrDefaultAsync(i => i.IngredientId == id);

            if (ingredient == null)
                return false;

            ingredient.IsActive = isActive;

            await context.SaveChangesAsync();

            return true;
        }
    }

    public class OrderRepository
    {
        private readonly DbContextOptions<CoffeeRuuDbContext> _options;

        public OrderRepository(
            DbContextOptions<CoffeeRuuDbContext> options)
        {
            _options = options;
        }

        public CoffeeRuuDbContext CreateContext()
        {
            return new CoffeeRuuDbContext(_options);
        }

        // ================================
        // LẤY MENU
        // ================================
        public async Task<List<MenuChoice>> GetMenuAsync()
        {
            await using var db = CreateContext();

            return await db.ProductSizes
                .AsNoTracking()
                .Where(s =>
                    s.IsAvailable &&
                    s.Product.IsAvailable &&
                    s.Product.Category.IsActive)
                .OrderBy(s => s.Product.ProductName)
                .ThenBy(s => s.SizeName)
                .Select(s => new MenuChoice
                {
                    ProductSizeId = s.ProductSizeId,
                    ProductName = s.Product.ProductName,
                    SizeName = s.SizeName,
                    Price = s.Price
                })
                .ToListAsync();
        }

        // ================================
        // LẤY DANH SÁCH BÀN
        // ================================
        public async Task<List<TableChoice>> GetTablesAsync()
        {
            await using var db = CreateContext();

            return await db.CafeTables
                .AsNoTracking()
                .Where(t =>
                    t.IsActive &&
                    (t.TableStatus == "Available" ||
                     t.TableStatus == "Occupied"))
                .OrderBy(t => t.TableName)
                .Select(t => new TableChoice
                {
                    TableId = t.TableId,
                    TableName = t.TableName
                })
                .ToListAsync();
        }

        // ================================
        // LẤY DANH SÁCH ĐƠN
        // ================================
        public async Task<List<OrderListRow>> GetOrdersAsync()
        {
            await using var db = CreateContext();

            return await db.Orders
                .AsNoTracking()
                .OrderByDescending(o => o.CreatedAt)
                .ThenByDescending(o => o.OrderId)
                .Take(100)
                .Select(o => new OrderListRow
                {
                    OrderId = o.OrderId,
                    OrderCode = o.OrderCode,
                    TableName = o.Table != null
                        ? o.Table.TableName
                        : "Mang đi",
                    OrderType = o.OrderType,
                    OrderStatus = o.OrderStatus,
                    TotalAmount = o.TotalAmount,
                    CreatedAt = o.CreatedAt
                })
                .ToListAsync();
        }

        // ================================
        // LẤY MÓN CHO KDS
        // ================================
        public async Task<List<KdsItemRow>> GetKdsAsync()
        {
            await using var db = CreateContext();

            return await db.OrderItems
                .AsNoTracking()
                .Where(i =>
                    (i.Order.OrderStatus == "New" ||
                     i.Order.OrderStatus == "Preparing" ||
                     i.Order.OrderStatus == "Ready") &&
                    (i.PreparationStatus == "Queued" ||
                     i.PreparationStatus == "Preparing" ||
                     i.PreparationStatus == "Ready"))
                .OrderBy(i => i.Order.CreatedAt)
                .ThenBy(i => i.OrderItemId)
                .Select(i => new KdsItemRow
                {
                    OrderItemId = i.OrderItemId,
                    OrderId = i.OrderId,
                    OrderCode = i.Order.OrderCode,

                    TableName = i.Order.Table != null
                        ? i.Order.Table.TableName
                        : "Mang đi",

                    ProductName = i.ProductNameSnapshot,
                    SizeName = i.SizeNameSnapshot,
                    Quantity = i.Quantity,
                    Note = i.Note ?? "",
                    PreparationStatus = i.PreparationStatus,
                    CreatedAt = i.Order.CreatedAt
                })
                .ToListAsync();
        }
    }

    public class InventoryService
    {
        private readonly IngredientRepository _ingredientRepository;

        public InventoryService(
            IngredientRepository ingredientRepository)
        {
            _ingredientRepository = ingredientRepository;
        }

        // 1. Lấy danh sách nguyên liệu
        public async Task<List<Ingredient>> GetIngredientsAsync()
        {
            return await _ingredientRepository.GetAllAsync();
        }

        // 2. Thêm nguyên liệu mới
        public async Task AddIngredientAsync(
            string name,
            string unit,
            decimal minimumStock)
        {
            ValidateIngredient(name, unit, minimumStock);

            name = name.Trim();
            unit = unit.Trim();

            bool exists = await _ingredientRepository
                .NameExistsAsync(name);

            if (exists)
                throw new InvalidOperationException(
                    "Tên nguyên liệu đã tồn tại.");

            var ingredient = new Ingredient
            {
                IngredientName = name,
                Unit = unit,
                CurrentStock = 0,
                MinimumStock = minimumStock,
                IsActive = true
            };

            try
            {
                await _ingredientRepository.AddAsync(ingredient);
            }
            catch (DbUpdateException ex)
                when (IsDuplicateKeyError(ex))
            {
                throw new InvalidOperationException(
                    "Tên nguyên liệu đã tồn tại.", ex);
            }
        }

        // 3. Cập nhật nguyên liệu
        public async Task UpdateIngredientAsync(
            int id,
            string name,
            string unit,
            decimal minimumStock)
        {
            ValidateIngredient(name, unit, minimumStock);

            name = name.Trim();
            unit = unit.Trim();

            var ingredient = await _ingredientRepository
                .GetByIdAsync(id);

            if (ingredient == null)
                throw new KeyNotFoundException(
                    "Không tìm thấy nguyên liệu.");

            bool exists = await _ingredientRepository
                .NameExistsAsync(name, id);

            if (exists)
                throw new InvalidOperationException(
                    "Tên nguyên liệu đã tồn tại.");

            try
            {
                bool updated = await _ingredientRepository
                    .UpdateInfoAsync(id, name, unit, minimumStock);

                if (!updated)
                    throw new KeyNotFoundException(
                        "Không tìm thấy nguyên liệu.");
            }
            catch (DbUpdateException ex)
                when (IsDuplicateKeyError(ex))
            {
                throw new InvalidOperationException(
                    "Tên nguyên liệu đã tồn tại.", ex);
            }
        }

        // 4. Kích hoạt hoặc ngừng sử dụng nguyên liệu
        public async Task SetIngredientActiveAsync(
            int id, bool isActive)
        {
            bool updated = await _ingredientRepository
                .SetActiveAsync(id, isActive);

            if (!updated)
                throw new KeyNotFoundException(
                    "Không tìm thấy nguyên liệu.");
        }

        // 5. Kiểm tra dữ liệu đầu vào
        private void ValidateIngredient(
            string name,
            string unit,
            decimal minimumStock)
        {
            if (string.IsNullOrWhiteSpace(name))
                throw new ArgumentException(
                    "Tên nguyên liệu không được để trống.");

            if (name.Trim().Length > 120)
                throw new ArgumentException(
                    "Tên nguyên liệu không được vượt quá 120 ký tự.");

            if (string.IsNullOrWhiteSpace(unit))
                throw new ArgumentException(
                    "Đơn vị tính không được để trống.");

            if (unit.Trim().Length > 30)
                throw new ArgumentException(
                    "Đơn vị tính không được vượt quá 30 ký tự.");

            if (minimumStock < 0)
                throw new ArgumentException(
                    "Tồn kho tối thiểu không được âm.");

            if (minimumStock > 999999999.999m ||
                decimal.Round(minimumStock, 3) != minimumStock)
                throw new ArgumentException(
                    "Tồn kho tối thiểu phải phù hợp decimal(12,3).");
        }

        // 6. Nhận diện lỗi trùng khóa từ SQL Server
        private static bool IsDuplicateKeyError(
            DbUpdateException exception)
        {
            return exception.InnerException is SqlException sqlEx
                && (sqlEx.Number == 2601 ||
                    sqlEx.Number == 2627);
        }
    }

        // ==========================

        public async Task<Order> CreateOrderAsync(
            IReadOnlyCollection<CartLine> cart,
            string orderType,
            int? tableId,
            int employeeId,
            string? note)
        {
            if (cart == null || cart.Count == 0)
                throw new InvalidOperationException(
                    "Giỏ hàng đang trống.");

            if (orderType != "DineIn" &&
                orderType != "Takeaway")
                throw new InvalidOperationException(
                    "Loại đơn không hợp lệ.");

            if (orderType == "DineIn" && !tableId.HasValue)
                throw new InvalidOperationException(
                    "Vui lòng chọn bàn.");

            if ((note?.Length ?? 0) > 500)
                throw new InvalidOperationException(
                    "Ghi chú đơn tối đa 500 ký tự.");

            if (cart.Any(i =>
                i.Quantity <= 0 ||
                i.Quantity > 100 ||
                (i.Note?.Length ?? 0) > 300))
            {
                throw new InvalidOperationException(
                    "Số lượng hoặc ghi chú món không hợp lệ.");
            }

            if (orderType == "Takeaway")
                tableId = null;

            await using var db =
                _repository.CreateContext();

            await using var transaction =
                await db.Database.BeginTransactionAsync(
                    IsolationLevel.Serializable);

            await CheckEmployeeAsync(db, employeeId);

            // Kiểm tra bàn
            if (tableId.HasValue)
            {
                bool tableValid = await db.CafeTables
                    .AnyAsync(t =>
                        t.TableId == tableId.Value &&
                        t.IsActive &&
                        (t.TableStatus == "Available" ||
                         t.TableStatus == "Occupied"));

                if (!tableValid)
                    throw new InvalidOperationException(
                        "Bàn không hợp lệ hoặc ngừng hoạt động.");
            }

            // Lấy giá và sản phẩm thực tế
            var productSizeIds = cart
                .Select(i => i.ProductSizeId)
                .Distinct()
                .ToList();

            var products = await db.ProductSizes
                .AsNoTracking()
                .Where(s =>
                    productSizeIds.Contains(s.ProductSizeId) &&
                    s.IsAvailable &&
                    s.Product.IsAvailable &&
                    s.Product.Category.IsActive)
                .Select(s => new
                {
                    s.ProductSizeId,
                    s.SizeName,
                    s.Price,
                    s.Product.ProductName
                })
                .ToDictionaryAsync(s => s.ProductSizeId);

            if (products.Count != productSizeIds.Count)
                throw new InvalidOperationException(
                    "Có sản phẩm không còn bán. Hãy làm mới menu.");

            var now = DateTime.Now;

            var order = new Order
            {
                OrderCode = "CR" +
                    Guid.NewGuid().ToString("N")[..18],

                TableId = tableId,
                CreatedBy = employeeId,
                OrderType = orderType,
                OrderStatus = "New",

                Subtotal = 0,
                DiscountAmount = 0,
                TotalAmount = 0,

                Note = note?.Trim(),
                CreatedAt = now
            };

            // Thêm món vào đơn
            foreach (var cartItem in cart)
            {
                var product =
                    products[cartItem.ProductSizeId];

                decimal lineTotal =
                    product.Price * cartItem.Quantity;

                order.OrderItems.Add(new OrderItem
                {
                    ProductSizeId =
                        cartItem.ProductSizeId,

                    ProductNameSnapshot =
                        product.ProductName,

                    SizeNameSnapshot =
                        product.SizeName,

                    UnitPrice = product.Price,
                    Quantity = cartItem.Quantity,
                    LineTotal = lineTotal,

                    PreparationStatus = "Queued",
                    Note = (cartItem.Note ?? "").Trim()
                });
            }

            // Tính tổng tiền
            order.Subtotal = order.OrderItems
                .Sum(i => i.LineTotal);

            order.TotalAmount =
                order.Subtotal - order.DiscountAmount;

            // Lưu lịch sử tạo đơn
            order.OrderStatusHistories.Add(
                new OrderStatusHistory
                {
                    ChangedBy = employeeId,
                    OldStatus = null,
                    NewStatus = "New",
                    ChangedAt = now,
                    Note = "Tạo đơn hàng"
                });

            db.Orders.Add(order);

            await db.SaveChangesAsync();
            await transaction.CommitAsync();

            return order;
        }

        // ==========================
        // 2. KDS: CẬP NHẬT PHA CHẾ
        // ==========================

        public async Task UpdatePreparationAsync(
            int orderItemId,
            string nextStatus,
            int employeeId)
        {
            if (nextStatus != "Preparing" &&
                nextStatus != "Ready")
            {
                throw new InvalidOperationException(
                    "Trạng thái pha chế không hợp lệ.");
            }

            await using var db =
                _repository.CreateContext();

            // Xác định đơn hàng chứa món
            int? orderId = await db.OrderItems
                .AsNoTracking()
                .Where(i => i.OrderItemId == orderItemId)
                .Select(i => (int?)i.OrderId)
                .FirstOrDefaultAsync();

            if (!orderId.HasValue)
                throw new InvalidOperationException(
                    "Không tìm thấy món.");

            await using var transaction =
                await db.Database.BeginTransactionAsync(
                    IsolationLevel.Serializable);

            await CheckEmployeeAsync(db, employeeId);
            await LockOrderAsync(db, orderId.Value);

            var order = await db.Orders
                .Include(o => o.OrderItems)
                .SingleOrDefaultAsync(o =>
                    o.OrderId == orderId.Value);

            if (order == null)
                throw new InvalidOperationException(
                    "Không tìm thấy đơn hàng.");

            if (order.OrderStatus != "New" &&
                order.OrderStatus != "Preparing" &&
                order.OrderStatus != "Ready")
            {
                throw new InvalidOperationException(
                    "Đơn hàng không còn được pha chế.");
            }

            var item = order.OrderItems
                .SingleOrDefault(i =>
                    i.OrderItemId == orderItemId);

            if (item == null)
                throw new InvalidOperationException(
                    "Món không thuộc đơn hàng này.");

            // Chỉ cho chuyển theo thứ tự
            bool isValid =
                (item.PreparationStatus == "Queued" &&
                 nextStatus == "Preparing") ||
                (item.PreparationStatus == "Preparing" &&
                 nextStatus == "Ready");

            if (!isValid)
                throw new InvalidOperationException(
                    $"Không thể chuyển từ " +
                    $"{item.PreparationStatus} sang {nextStatus}.");

            item.PreparationStatus = nextStatus;

            // Tính lại trạng thái tổng của đơn
            bool allReady = order.OrderItems
                .All(i => i.PreparationStatus == "Ready");

            string newOrderStatus = allReady
                ? "Ready"
                : "Preparing";

            ChangeOrderStatus(
                db,
                order,
                newOrderStatus,
                employeeId,
                "Cập nhật tiến độ pha chế");

            await db.SaveChangesAsync();
            await transaction.CommitAsync();
        }

        // ==========================
        // 3. BÀN GIAO ĐƠN
        // ==========================

        public async Task CompleteOrderAsync(
            int orderId,
            int employeeId)
        {
            await using var db =
                _repository.CreateContext();

            await using var transaction =
                await db.Database.BeginTransactionAsync(
                    IsolationLevel.Serializable);

            await CheckEmployeeAsync(db, employeeId);
            await LockOrderAsync(db, orderId);

            var order = await db.Orders
                .Include(o => o.OrderItems)
                .SingleOrDefaultAsync(o =>
                    o.OrderId == orderId);

            if (order == null)
                throw new InvalidOperationException(
                    "Không tìm thấy đơn hàng.");

            if (order.OrderStatus != "Ready")
                throw new InvalidOperationException(
                    "Đơn chưa sẵn sàng để bàn giao.");

            if (order.OrderItems.Count == 0 ||
                order.OrderItems.Any(i =>
                    i.PreparationStatus != "Ready"))
            {
                throw new InvalidOperationException(
                    "Vẫn còn món chưa pha xong.");
            }

            // Cập nhật các món đã giao
            foreach (var item in order.OrderItems)
            {
                item.PreparationStatus = "Served";
            }

            order.CompletedAt = DateTime.Now;

            ChangeOrderStatus(
                db,
                order,
                "Completed",
                employeeId,
                "Đã bàn giao đơn hàng");

            await db.SaveChangesAsync();
            await transaction.CommitAsync();
        }

        public async Task CancelNewOrderAsync(
            int orderId,
            int employeeId)
        {
            await using var db =
                _repository.CreateContext();

            await using var transaction =
                await db.Database.BeginTransactionAsync(
                    IsolationLevel.Serializable);

            await CheckEmployeeAsync(db, employeeId);
            await LockOrderAsync(db, orderId);

            var order = await db.Orders
                .Include(o => o.OrderItems)
                .SingleOrDefaultAsync(o =>
                    o.OrderId == orderId);

            if (order == null)
                throw new InvalidOperationException(
                    "Không tìm thấy đơn hàng.");

            if (order.OrderStatus != "New" ||
                order.OrderItems.Any(i =>
                    i.PreparationStatus != "Queued"))
            {
                throw new InvalidOperationException(
                    "Chỉ được hủy đơn chưa pha chế.");
            }

            // Không hủy đơn đã có giao dịch thanh toán
            bool hasPayment = await db.Payments
                .AnyAsync(p => p.OrderId == orderId);

            if (hasPayment)
                throw new InvalidOperationException(
                    "Đơn có giao dịch thanh toán. " +
                    "Cần bộ phận Payment xử lý.");

            foreach (var item in order.OrderItems)
            {
                item.PreparationStatus = "Cancelled";
            }

            ChangeOrderStatus(
                db,
                order,
                "Cancelled",
                employeeId,
                "Hủy đơn chưa pha chế");

            await db.SaveChangesAsync();
            await transaction.CommitAsync();
        }

        // ==========================
        // 5. KIỂM TRA NHÂN VIÊN
        // ==========================

        private static async Task CheckEmployeeAsync(
            CoffeeRuuDbContext db,
            int employeeId)
        {
            bool valid = employeeId > 0 &&
                await db.Employees.AnyAsync(e =>
                    e.EmployeeId == employeeId &&
                    e.IsActive);

            if (!valid)
                throw new InvalidOperationException(
                    "EmployeeId không hợp lệ hoặc ngừng hoạt động.");
        }

        // ==========================
        // 6. KHÓA ĐƠN KHI CẬP NHẬT
        // ==========================

        private static async Task LockOrderAsync(
            CoffeeRuuDbContext db,
            int orderId)
        {
            var connection =
                db.Database.GetDbConnection();

            await using var command =
                connection.CreateCommand();

            command.Transaction =
                db.Database.CurrentTransaction!
                    .GetDbTransaction();

            command.CommandText = @"
                SELECT OrderId
                FROM dbo.Orders WITH (UPDLOCK, HOLDLOCK)
                WHERE OrderId = @id";

            var parameter = command.CreateParameter();
            parameter.ParameterName = "@id";
            parameter.Value = orderId;

            command.Parameters.Add(parameter);

            var result =
                await command.ExecuteScalarAsync();

            if (result == null || result == DBNull.Value)
                throw new InvalidOperationException(
                    "Không tìm thấy đơn hàng.");
        }

        // ==========================
        // 7. LƯU LỊCH SỬ TRẠNG THÁI
        // ==========================

        private static void ChangeOrderStatus(
            CoffeeRuuDbContext db,
            Order order,
            string nextStatus,
            int employeeId,
            string note)
        {
            string oldStatus = order.OrderStatus;

            if (oldStatus == nextStatus)
                return;

            order.OrderStatus = nextStatus;

            db.OrderStatusHistories.Add(
                new OrderStatusHistory
                {
                    OrderId = order.OrderId,
                    ChangedBy = employeeId,
                    OldStatus = oldStatus,
                    NewStatus = nextStatus,
                    ChangedAt = DateTime.Now,
                    Note = note
                });
        }

using System;
using System.Security.Cryptography;

namespace CoffeeRuu.Services
{
    public class PasswordService
    {
        public static string HashPassword(string password)
        {
            byte[] salt = RandomNumberGenerator.GetBytes(16);

            byte[] hash = Rfc2898DeriveBytes.Pbkdf2(
                password,
                salt,
                100000,
                HashAlgorithmName.SHA256,
                32);

            return $"{Convert.ToBase64String(salt)}:{Convert.ToBase64String(hash)}";
        }

        public static bool VerifyPassword(
            string password,
            string storedHash)
        {
            try
            {
                string[] parts = storedHash.Split(':');

                if (parts.Length != 2)
                    return false;

                byte[] salt = Convert.FromBase64String(parts[0]);
                byte[] expectedHash = Convert.FromBase64String(parts[1]);

                byte[] actualHash = Rfc2898DeriveBytes.Pbkdf2(
                    password,
                    salt,
                    100000,
                    HashAlgorithmName.SHA256,
                    32);

                return CryptographicOperations.FixedTimeEquals(
                    actualHash,
                    expectedHash);
            }
            catch (FormatException)
            {
                return false;
            }
        }
    }
}

            string? address)
        {
            ValidateSupplier(name, phone, email, address);

            var supplier = new Supplier
            {
                SupplierName = name.Trim(),
                Phone = NormalizeOptional(phone),
                Email = NormalizeOptional(email),
                Address = NormalizeOptional(address),
                IsActive = true
            };

            await _supplierRepository.AddAsync(supplier);
        }

        // 4. Cập nhật nhà cung cấp
        public async Task UpdateSupplierAsync(
            int id,
            string name,
            string? phone,
            string? email,
            string? address)
        {
            ValidateSupplier(name, phone, email, address);

            bool updated = await _supplierRepository.UpdateInfoAsync(
                id,
                name.Trim(),
                NormalizeOptional(phone),
                NormalizeOptional(email),
                NormalizeOptional(address)
            );

            if (!updated)
                throw new KeyNotFoundException(
                    "Không tìm thấy nhà cung cấp.");
        }

        // 5. Kích hoạt hoặc ngừng hợp tác
        public async Task SetSupplierActiveAsync(
            int id, bool isActive)
        {
            bool updated = await _supplierRepository
                .SetActiveAsync(id, isActive);

            if (!updated)
                throw new KeyNotFoundException(
                    "Không tìm thấy nhà cung cấp.");
        }

        // 6. Kiểm tra dữ liệu đầu vào
        private void ValidateSupplier(
            string name,
            string? phone,
            string? email,
            string? address)
        {
            if (string.IsNullOrWhiteSpace(name))
                throw new ArgumentException(
                    "Tên nhà cung cấp không được để trống.");

            if (name.Trim().Length > 150)
                throw new ArgumentException(
                    "Tên nhà cung cấp tối đa 150 ký tự.");

            if (!string.IsNullOrWhiteSpace(phone)
                && phone.Trim().Length > 20)
                throw new ArgumentException(
                    "Số điện thoại tối đa 20 ký tự.");

            if (!string.IsNullOrWhiteSpace(email))
            {
                email = email.Trim();

                if (email.Length > 150)
                    throw new ArgumentException(
                        "Email tối đa 150 ký tự.");

                if (!MailAddress.TryCreate(email, out var mail)
                    || !string.Equals(mail.Address, email,
                        StringComparison.OrdinalIgnoreCase))
                    throw new ArgumentException(
                        "Email không đúng định dạng.");
            }

            if (!string.IsNullOrWhiteSpace(address)
                && address.Trim().Length > 300)
                throw new ArgumentException(
                    "Địa chỉ tối đa 300 ký tự.");
        }

        // 7. Chuẩn hóa thông tin không bắt buộc
        private static string? NormalizeOptional(string? value)
        {
            if (string.IsNullOrWhiteSpace(value))
                return null;

            return value.Trim();
        }
    }

            bool exists = await _tableRepository.NameExistsAsync(name.Trim(), id);
            if (exists) throw new InvalidOperationException("Tên bàn đã tồn tại ở một bàn khác.");

        {
            if (string.IsNullOrWhiteSpace(status))

        {

                throw new ArgumentException("Tên bàn tối đa 50 ký tự.");

            if (capacity <= 0)
                throw new ArgumentException("Sức chứa (số ghế) phải lớn hơn 0.");
        }

                TableName = tableName.Trim(),
                Capacity = capacity,

            {
                throw new KeyNotFoundException("Không tìm thấy bàn cần xóa.");
            }
        }

                return;

            if (string.IsNullOrWhiteSpace(Password))
            {
                MessageBox.Show("Vui long nhap mat khau!");
                return;

            try
            {
                using var db = App.CreateDbContext();

                var employee = db.Employees
                    .FirstOrDefault(e => e.Username == Username);

                    !employee.IsActive ||

                foreach (Window window in Application.Current.Windows)
                {
                    if (window is LoginView)
                    {
                        window.Close();
                        break;
                    }
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    ex.Message,
                    "Loi dang nhap",
                    MessageBoxButton.OK,

namespace CoffeeRuu.ViewModels
{
    public abstract class NotifyBase : INotifyPropertyChanged
    {
        public event PropertyChangedEventHandler? PropertyChanged;

        protected bool Set<T>(
            ref T field,
            T value,
            [CallerMemberName] string? name = null)
        {
            if (EqualityComparer<T>.Default.Equals(field, value))
                return false;

            field = value;

            PropertyChanged?.Invoke(
                this,
                new PropertyChangedEventArgs(name));

            return true;
        }

        protected void Notify(
            [CallerMemberName] string? name = null)
        {
            PropertyChanged?.Invoke(
                this,
                new PropertyChangedEventArgs(name));
        }
    }

namespace CoffeeRuu.ViewModels

    public class RelayCommand : ICommand
    {
        private readonly Action _execute;
        private readonly Func<bool>? _canExecute;

        public RelayCommand(
            Action execute,
            Func<bool>? canExecute = null)
        {
            _execute = execute;
            _canExecute = canExecute;
        }

        public bool CanExecute(object? parameter)
        {
            return _canExecute == null || _canExecute();
        }

        public void Execute(object? parameter)
        {
            _execute();
        }

        public event EventHandler? CanExecuteChanged;

        public void RaiseCanExecuteChanged()
        {
            CanExecuteChanged?.Invoke(this, EventArgs.Empty);
        }
    }

        // Commands
        public RelayCommand LoadCommand { get; }
        public RelayCommand AddCommand { get; }
        public RelayCommand UpdateCommand { get; }
        public RelayCommand ToggleActiveCommand { get; }
        public RelayCommand ClearCommand { get; }

        // Constructor
        public SupplierViewModel(
            SupplierService supplierService)
        {
            _supplierService = supplierService;

            SuppliersView =
                CollectionViewSource.GetDefaultView(Suppliers);

            SuppliersView.Filter = FilterSupplier;

            LoadCommand = new RelayCommand(
                async () => await ExecuteSafeAsync(
                    LoadSuppliersAsync),
                () => !IsBusy);

            AddCommand = new RelayCommand(
                async () => await ExecuteSafeAsync(
                    AddSupplierAsync),
                () => !IsBusy && SelectedSupplier == null);

            UpdateCommand = new RelayCommand(
                async () => await ExecuteSafeAsync(
                    UpdateSupplierAsync),
                () => !IsBusy && SelectedSupplier != null);

            ToggleActiveCommand = new RelayCommand(
                async () => await ExecuteSafeAsync(
                    ToggleActiveAsync),
                () => !IsBusy && SelectedSupplier != null);

            ClearCommand = new RelayCommand(
                ClearForm,
                () => !IsBusy);

            // Tải dữ liệu khi tạo ViewModel
            _ = ExecuteSafeAsync(LoadSuppliersAsync);
        }

        // 1. Lấy danh sách nhà cung cấp
        private async Task LoadSuppliersAsync()
        {
            int? selectedId = SelectedSupplier?.SupplierId;

            var items = await _supplierService
                .GetSuppliersAsync();

            Suppliers.Clear();

            foreach (var item in items)
            {
                Suppliers.Add(item);
            }

            SelectedSupplier = selectedId.HasValue
                ? Suppliers.FirstOrDefault(
                    s => s.SupplierId == selectedId.Value)
                : null;

            SuppliersView.Refresh();
        }

        // 2. Thêm nhà cung cấp
        private async Task AddSupplierAsync()
        {
            await _supplierService.AddSupplierAsync(
                SupplierNameInput,
                PhoneInput,
                EmailInput,
                AddressInput);

            await LoadSuppliersAsync();

            ClearForm();

            StatusMessage = "Thêm nhà cung cấp thành công.";
        }

        // 3. Cập nhật nhà cung cấp
        private async Task UpdateSupplierAsync()
        {
            if (SelectedSupplier == null)
                return;

            int id = SelectedSupplier.SupplierId;

            await _supplierService.UpdateSupplierAsync(
                id,
                SupplierNameInput,
                PhoneInput,
                EmailInput,
                AddressInput);

            await LoadSuppliersAsync();

            StatusMessage = "Cập nhật nhà cung cấp thành công.";
        }

        // 4. Kích hoạt / ngừng hợp tác
        private async Task ToggleActiveAsync()
        {
            if (SelectedSupplier == null)
                return;

            bool newStatus = !SelectedSupplier.IsActive;

            await _supplierService.SetSupplierActiveAsync(
                SelectedSupplier.SupplierId,
                newStatus);

            await LoadSuppliersAsync();

            StatusMessage = newStatus
                ? "Đã kích hoạt nhà cung cấp."
                : "Đã ngừng hợp tác với nhà cung cấp.";
        }

        // 5. Lọc danh sách nhà cung cấp
        private bool FilterSupplier(object obj)
        {
            if (obj is not Supplier supplier)
                return false;

            if (string.IsNullOrWhiteSpace(SearchText))
                return true;

            string keyword = SearchText.Trim();

            return supplier.SupplierName.Contains(
                       keyword,
                       StringComparison.OrdinalIgnoreCase)
                   || (supplier.Phone?.Contains(
                       keyword,
                       StringComparison.OrdinalIgnoreCase) ?? false)
                   || (supplier.Email?.Contains(
                       keyword,
                       StringComparison.OrdinalIgnoreCase) ?? false);
        }

        // 6. Xóa thông tin nhập
        private void ClearInputs()
        {
            SupplierNameInput = "";
            PhoneInput = "";
            EmailInput = "";
            AddressInput = "";
        }

        private void ClearForm()
        {
            SelectedSupplier = null;
            ClearInputs();
        }

        // 7. Xử lý thao tác bất đồng bộ
        private async Task ExecuteSafeAsync(Func<Task> action)
        {
            if (IsBusy)
                return;

            try
            {
                IsBusy = true;
                StatusMessage = "";

                await action();
            }
            catch (ArgumentException ex)
            {
                StatusMessage = ex.Message;
            }
            catch (InvalidOperationException ex)
            {
                StatusMessage = ex.Message;
            }
            catch (KeyNotFoundException ex)
            {
                StatusMessage = ex.Message;
            }
            catch (Exception ex)
            {
                StatusMessage =
                    "Có lỗi xảy ra: " + ex.Message;
            }
            finally
            {
                IsBusy = false;
            }
        }

        // 8. Cập nhật trạng thái các Command
        private void UpdateCommands()
        {
            LoadCommand?.RaiseCanExecuteChanged();
            AddCommand?.RaiseCanExecuteChanged();
            UpdateCommand?.RaiseCanExecuteChanged();
            ToggleActiveCommand?.RaiseCanExecuteChanged();
            ClearCommand?.RaiseCanExecuteChanged();
        }

        // 9. Thông báo thay đổi thuộc tính
        public event PropertyChangedEventHandler?
            PropertyChanged;

        private void OnPropertyChanged(
            [CallerMemberName] string? propertyName = null)
        {
            PropertyChanged?.Invoke(
                this,
                new PropertyChangedEventArgs(propertyName));
        }
    }

    public class TableViewModel : INotifyPropertyChanged
    {
        private readonly TableService _tableService;

        public ObservableCollection<CafeTable> Tables { get; } = new();

        public ObservableCollection<string> StatusList { get; } = new()
        {
            "Available", "Occupied", "Reserved", "OutOfService"
        };

        private CafeTable? _selectedTable;
        public CafeTable? SelectedTable
        {
            get => _selectedTable;
            set
            {
                if (_selectedTable == value) return;

                _selectedTable = value;
                OnPropertyChanged();

                if (value != null)
                {
                    TableNameInput = value.TableName;
                    CapacityInput = value.Capacity.ToString();
                    StatusInput = value.TableStatus;
                }
                else
                {
                    TableNameInput = GetNextTableName();
                    CapacityInput = "4";
                    StatusInput = "Available";
                    StatusMessage = "";
                }

                UpdateCommands();
            }
        }

        private string _tableNameInput = "";
        public string TableNameInput
        {
            get => _tableNameInput;
            set { _tableNameInput = value; OnPropertyChanged(); }
        }

        private string _capacityInput = "4";
        public string CapacityInput
        {
            get => _capacityInput;
            set { _capacityInput = value; OnPropertyChanged(); }
        }

        private string _statusInput = "Available";
        public string StatusInput
        {
            get => _statusInput;
            set { _statusInput = value; OnPropertyChanged(); }
        }

        private string _statusMessage = "";
        public string StatusMessage
        {
            get => _statusMessage;
            set { _statusMessage = value; OnPropertyChanged(); }
        }

        public RelayCommand AddCommand { get; }
        public RelayCommand LoadCommand { get; }
        public RelayCommand UpdateCommand { get; }
        public RelayCommand DeleteCommand { get; }
        public RelayCommand ClearCommand { get; }

        public TableViewModel(TableService tableService)
        {
            _tableService = tableService;

            // Nút Thêm chỉ sáng lên khi KHÔNG CÓ bàn nào được chọn (đang ở chế độ tạo mới)
            AddCommand = new RelayCommand(async () => await AddTableAsync(), () => SelectedTable == null);

            // Nút Cập nhật chỉ sáng lên khi CÓ bàn được chọn
            UpdateCommand = new RelayCommand(async () => await UpdateTableAsync(), () => SelectedTable != null);

            // ĐÃ THÊM: Nút Xóa chỉ sáng lên khi CÓ bàn được chọn
            DeleteCommand = new RelayCommand(async () => await DeleteTableAsync(), () => SelectedTable != null);

            ClearCommand = new RelayCommand(ClearInputs);
            LoadCommand = new RelayCommand(async () => await LoadTablesAsync());

            _ = LoadTablesAsync();
        }

        // --- Hàm tự động tính Tên bàn tiếp theo (Định dạng TXX) ---
        private string GetNextTableName()
        {
            if (Tables == null || Tables.Count == 0) return "T01";

            int maxNumber = 0;
            foreach (var table in Tables)
            {
                string numbers = new string(table.TableName.Where(char.IsDigit).ToArray());
                if (int.TryParse(numbers, out int num))
                {
                    if (num > maxNumber) maxNumber = num;
                }
            }

            return $"T{(maxNumber + 1):D2}";
        }

        // --- Hàm xử lý Thêm bàn ---
        private async Task AddTableAsync()
        {
            try
            {
                if (!int.TryParse(CapacityInput, out int capacity))
                    throw new ArgumentException("Sức chứa phải là số nguyên.");

                string newTableName = GetNextTableName(); // Tính lại tên cho chắc chắn trước khi lưu

                await _tableService.AddTableAsync(newTableName, capacity, StatusInput);

                await LoadTablesAsync();
                StatusMessage = $"Thêm {newTableName} thành công!";
                ClearInputs(); // Reset form sau khi thêm
            }
            catch (Exception ex) { StatusMessage = ex.Message; }
        }

        private async Task LoadTablesAsync()
        {
            var items = await _tableService.GetTablesAsync();
            Tables.Clear();
            foreach (var item in items)
            {
                Tables.Add(item);
            }

            if (SelectedTable == null)
            {
                TableNameInput = GetNextTableName();
            }
        }

        private async Task UpdateTableAsync()
        {
            try
            {
                if (SelectedTable == null) return;
                if (!int.TryParse(CapacityInput, out int capacity))
                    throw new ArgumentException("Sức chứa phải là số nguyên.");

                await _tableService.UpdateTableAsync(SelectedTable.TableId, TableNameInput, capacity);
                await _tableService.ChangeTableStatusAsync(SelectedTable.TableId, StatusInput);

                await LoadTablesAsync();
                StatusMessage = "Cập nhật bàn thành công!";
            }
            catch (Exception ex) { StatusMessage = ex.Message; }
        }

        // ĐÃ THÊM: Hàm xử lý Xóa bàn
        private async Task DeleteTableAsync()
        {
            try
            {
                if (SelectedTable == null) return;

                await _tableService.DeleteTableAsync(SelectedTable.TableId);

                await LoadTablesAsync();
                ClearInputs(); // Reset form sau khi xóa
                StatusMessage = "Xóa bàn thành công!";
            }
            catch (Exception ex) { StatusMessage = ex.Message; }
        }

        private void ClearInputs()
        {
            _selectedTable = null;
            OnPropertyChanged(nameof(SelectedTable));

            TableNameInput = GetNextTableName(); // Đặt sẵn tên bàn tiếp theo
            CapacityInput = "4";
            StatusInput = "Available";
            StatusMessage = "";

            UpdateCommands();
        }

        private void UpdateCommands()
        {
            AddCommand?.RaiseCanExecuteChanged();
            UpdateCommand?.RaiseCanExecuteChanged();

            DeleteCommand?.RaiseCanExecuteChanged();

            ClearCommand?.RaiseCanExecuteChanged();
        }

        public event PropertyChangedEventHandler? PropertyChanged;
        protected void OnPropertyChanged([CallerMemberName] string? propertyName = null)
        {
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
        }
    }

using System;
using System.Linq;
using System.Threading.Tasks;
using System.Windows;
using System.Windows.Controls;
using CoffeeRuu.Models;
using CoffeeRuu.Services;

namespace CoffeeRuu.Views
{
    public partial class MenuView : UserControl
    {
        private readonly MenuService _menuService;

        public MenuView()
            : this(new MenuService(
                new CoffeeRuu.Repositories.MenuRepository(
                    App.CreateDbContext())))
        {
        }

        public MenuView(MenuService menuService)
        {
            InitializeComponent();

            _menuService = menuService;
            Loaded += MenuView_Loaded;
        }

        private async void MenuView_Loaded(
            object sender, RoutedEventArgs e)
        {
            await LoadMenuAsync();
        }

        // READ: Tải danh sách món uống
        private async Task LoadMenuAsync()
        {
            try
            {
                MenuDataGrid.ItemsSource =
                    await _menuService.GetAllProductsAsync();
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Không thể tải thực đơn: " + ex.Message);
            }
        }

        // SEARCH: Tìm theo tên món
        private async void SearchButton_Click(
            object sender, RoutedEventArgs e)
        {
            try
            {
                var products =
                    await _menuService.GetAllProductsAsync();

                string keyword = SearchBox.Text.Trim();

                MenuDataGrid.ItemsSource = products
                    .Where(p => p.ProductName != null &&
                        p.ProductName.Contains(
                            keyword,
                            StringComparison.OrdinalIgnoreCase))
                    .ToList();
            }
            catch (Exception ex)
            {
                MessageBox.Show("Lỗi tìm kiếm: " + ex.Message);
            }
        }

        // CREATE: Thêm món uống
        private async void AddButton_Click(
            object sender, RoutedEventArgs e)
        {
            var dialog = new ProductDialog
            {
                Owner = Window.GetWindow(this)
            };

            if (dialog.ShowDialog() != true ||
                dialog.ProductResult == null)
                return;

            try
            {
                await _menuService.AddProductAsync(
                    dialog.ProductResult);

                await LoadMenuAsync();

                MessageBox.Show("Thêm món uống thành công!");
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Không thể thêm món: " + ex.Message);
            }
        }

        // UPDATE: Sửa món được chọn
        private async void EditButton_Click(
            object sender, RoutedEventArgs e)
        {
            if (MenuDataGrid.SelectedItem is not Product selected)
            {
                MessageBox.Show("Vui lòng chọn món cần sửa.");
                return;
            }

            var dialog = new ProductDialog(selected)
            {
                Owner = Window.GetWindow(this)
            };

            if (dialog.ShowDialog() != true ||
                dialog.ProductResult == null)
                return;

            try
            {
                await _menuService.UpdateProductAsync(
                    dialog.ProductResult);

                await LoadMenuAsync();

                MessageBox.Show("Cập nhật món uống thành công!");
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Không thể sửa món: " + ex.Message);
            }
        }

        // DELETE: Ẩn món khỏi thực đơn

        private async void DeleteButton_Click(
            object sender, RoutedEventArgs e)
        {
            if (MenuDataGrid.SelectedItem is not Product product)
            {
                MessageBox.Show("Vui lòng chọn món cần xóa.");
                return;
            }

            MessageBoxResult result = MessageBox.Show(
                $"Bạn có chắc chắn muốn xóa món '{product.ProductName}' không?\n" +
                "Thao tác này sẽ xóa bản ghi món khỏi database.",
                "Xác nhận xóa",
                MessageBoxButton.YesNo,
                MessageBoxImage.Warning);

            if (result != MessageBoxResult.Yes)
                return;

            try
            {
                await _menuService.DeleteProductAsync(product.ProductId);

                await LoadMenuAsync();

                MessageBox.Show("Đã xóa món uống thành công.");
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Không thể xóa món uống. Có thể món đang được sử dụng " +
                    "trong size, công thức hoặc đơn hàng.\n\nChi tiết: " +
                    ex.Message,
                    "Lỗi xóa",
                    MessageBoxButton.OK,
                    MessageBoxImage.Error);
            }
        }

    }
}

using System;
using System.Linq;
using System.Threading.Tasks;
using System.Windows;
using CoffeeRuu.Data;
using CoffeeRuu.Models;
using Microsoft.EntityFrameworkCore;

namespace CoffeeRuu.Views
{
    public partial class ProductDialog : Window
    {
        private readonly Product? _editingProduct;

        public Product? ProductResult { get; private set; }

        public ProductDialog(Product? product = null)
        {
            InitializeComponent();

            _editingProduct = product;

            if (_editingProduct != null)
            {
                TitleText.Text = "Sửa món uống";
                ProductNameBox.Text = _editingProduct.ProductName;
                DescriptionBox.Text = _editingProduct.Description ?? "";
                AvailableCheckBox.IsChecked = _editingProduct.IsAvailable;
            }

            Loaded += ProductDialog_Loaded;
        }

        private async void ProductDialog_Loaded(
            object sender, RoutedEventArgs e)
        {
            try
            {
                using var db = App.CreateDbContext();

                var categories = await db.Categories
                    .OrderBy(c => c.CategoryName)
                    .ToListAsync();

                CategoryComboBox.ItemsSource = categories;

                if (_editingProduct != null)
                {
                    CategoryComboBox.SelectedValue =
                        _editingProduct.CategoryId;
                }
                else if (categories.Count > 0)
                {
                    CategoryComboBox.SelectedIndex = 0;
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Không thể tải danh mục: " + ex.Message,
                    "Lỗi",
                    MessageBoxButton.OK,
                    MessageBoxImage.Error);
            }
        }

        private void SaveButton_Click(
            object sender, RoutedEventArgs e)
        {
            string name = ProductNameBox.Text.Trim();
            string description = DescriptionBox.Text.Trim();

            if (string.IsNullOrWhiteSpace(name))
            {
                MessageBox.Show("Vui lòng nhập tên món uống.");
                ProductNameBox.Focus();
                return;
            }

            if (CategoryComboBox.SelectedValue == null)
            {
                MessageBox.Show("Vui lòng chọn danh mục.");
                return;
            }

            int categoryId = Convert.ToInt32(
                CategoryComboBox.SelectedValue);

            ProductResult = new Product
            {
                ProductId = _editingProduct?.ProductId ?? 0,
                ProductName = name,
                CategoryId = categoryId,
                Description = string.IsNullOrWhiteSpace(description)
                    ? null
                    : description,
                ImageUrl = _editingProduct?.ImageUrl,
                IsAvailable = AvailableCheckBox.IsChecked == true,
                CreatedAt = _editingProduct?.CreatedAt ?? DateTime.Now
            };

            DialogResult = true;
        }

        private void CancelButton_Click(
            object sender, RoutedEventArgs e)
        {
            DialogResult = false;
        }
    }
}

        private void RefreshButton_Click(object sender, RoutedEventArgs e)
        {
            LoadReport();
        }

        private void LoadReport()
        {
            if (StartDatePicker.SelectedDate is not DateTime startDate ||
                EndDatePicker.SelectedDate is not DateTime endDate)
            {
                MessageBox.Show("Vui lòng chọn ngày bắt đầu và ngày kết thúc.",
                    "Thiếu ngày", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            if (startDate.Date > endDate.Date)
            {
                MessageBox.Show("Ngày bắt đầu không được sau ngày kết thúc.",
                    "Khoảng thời gian không hợp lệ", MessageBoxButton.OK, MessageBoxImage.Warning);
                return;
            }

            try
            {
                DateTime start = startDate.Date;
                DateTime endExclusive = endDate.Date.AddDays(1);

                using CoffeeRuuDbContext db = App.CreateDbContext();

                // Quy tắc nghiệp vụ: chỉ tính đơn Completed có ít nhất một thanh toán Success
                // và không có thanh toán Refunded. Ngày báo cáo = CompletedAt (nếu null thì CreatedAt).
                var validOrders = db.Orders
                    .AsNoTracking()
                    .Include(o => o.Payments)
                    .Include(o => o.OrderItems)
                    .Where(o => o.OrderStatus == "Completed"
                        && o.Payments.Any(p => p.PaymentStatus == "Success")
                        && !o.Payments.Any(p => p.PaymentStatus == "Refunded"))
                    .Where(o => (o.CompletedAt ?? o.CreatedAt) >= start
                        && (o.CompletedAt ?? o.CreatedAt) < endExclusive)
                    .ToList();

                decimal totalRevenue = validOrders.Sum(o => o.TotalAmount);
                int orderCount = validOrders.Count;
                decimal averageOrder = orderCount == 0 ? 0 : totalRevenue / orderCount;

                RevenueText.Text = totalRevenue.ToString("N0", ViCulture) + " ₫";
                OrderCountText.Text = orderCount.ToString("N0", ViCulture);
                AverageOrderText.Text = averageOrder.ToString("N0", ViCulture) + " ₫";

                _dailyRows = validOrders
                    .GroupBy(o => (o.CompletedAt ?? o.CreatedAt).Date)
                    .OrderBy(g => g.Key)
                    .Select(g => new DailyRevenueRow
                    {
                        Date = g.Key,
                        DateLabel = g.Key.ToString("dd/MM/yyyy"),
                        OrderCount = g.Count(),
                        Revenue = g.Sum(o => o.TotalAmount)
                    })
                    .ToList();

                // Sản phẩm bán chạy: gộp theo TÊN sản phẩm (mọi size), bỏ các món đã Cancelled.
                _productRows = validOrders
                    .SelectMany(o => o.OrderItems)
                    .Where(i => i.PreparationStatus != "Cancelled")
                    .GroupBy(i => i.ProductNameSnapshot)
                    .Select(g => new
                    {
                        ProductName = g.Key,
                        QuantitySold = g.Sum(i => i.Quantity),
                        Revenue = g.Sum(i => i.LineTotal)
                    })
                    .OrderByDescending(p => p.QuantitySold)
                    .ThenByDescending(p => p.Revenue)
                    .ThenBy(p => p.ProductName)
                    .Select((p, index) => new BestSellingProductRow
                    {
                        Rank = index + 1,
                        ProductName = p.ProductName,
                        QuantitySold = p.QuantitySold,
                        Revenue = p.Revenue
                    })
                    .ToList();

                DailyRevenueGrid.ItemsSource = _dailyRows;
                BestSellingGrid.ItemsSource = _productRows;

                _reportStart = start;
                _reportEnd = endDate.Date;

                StatusText.Text = $"Khoảng thời gian: {start:dd/MM/yyyy} - {endDate.Date:dd/MM/yyyy}. " +
                    "Chỉ tính đơn Completed có thanh toán Success (không có khoản Refunded). " +
                    $"Đã tìm thấy {orderCount} đơn hợp lệ.";
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "Không thể tải báo cáo. Hãy kiểm tra kết nối SQL Server và dữ liệu database.\n\n" + ex.Message,
                    "Lỗi tải báo cáo", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void ExportButton_Click(object sender, RoutedEventArgs e)
        {
            if (_reportStart == null || _reportEnd == null)
            {
                MessageBox.Show("Hãy bấm \"Lọc báo cáo\" để tải dữ liệu trước khi xuất.",
                    "Chưa có dữ liệu", MessageBoxButton.OK, MessageBoxImage.Information);
                return;
            }

            if (_dailyRows.Count == 0 && _productRows.Count == 0)
            {
                MessageBox.Show("Không có dữ liệu trong khoảng thời gian đã chọn để xuất.",
                    "Không có dữ liệu", MessageBoxButton.OK, MessageBoxImage.Information);
                return;
            }

            var dialog = new SaveFileDialog
            {
                Title = "Xuất báo cáo CoffeeRuu",
                Filter = "Excel Workbook (*.xlsx)|*.xlsx",
                FileName = $"CoffeeRuu_BaoCao_{_reportStart:yyyyMMdd}_{_reportEnd:yyyyMMdd}.xlsx",
                DefaultExt = ".xlsx",
                AddExtension = true
            };

            if (dialog.ShowDialog() != true)
                return;

            try
            {
                decimal totalRevenue = _dailyRows.Sum(r => r.Revenue);
                int totalOrders = _dailyRows.Sum(r => r.OrderCount);

                var rows = new List<XlsxRow>
                {
                    new(new object?[] { "COFFEERUU - BÁO CÁO DOANH THU" }, XlsxStyle.Bold),
                    new(new object?[] { "Từ ngày", _reportStart.Value.ToString("dd/MM/yyyy") }),
                    new(new object?[] { "Đến ngày", _reportEnd.Value.ToString("dd/MM/yyyy") }),
                    new(new object?[] { "Quy tắc", "Chỉ tính đơn Completed có thanh toán Success (không có khoản Refunded)" }),
                    new(new object?[] { "Tổng doanh thu (VND)", totalRevenue }),
                    new(new object?[] { "Tổng số đơn", totalOrders }),
                    new(new object?[] { }),
                    new(new object?[] { "DOANH THU THEO NGÀY" }, XlsxStyle.Bold),
                    new(new object?[] { "Ngày", "Số đơn", "Doanh thu (VND)" }, XlsxStyle.Bold)
                };

                foreach (var row in _dailyRows)
                    rows.Add(new XlsxRow(new object?[] { row.DateLabel, row.OrderCount, row.Revenue }));

                rows.Add(new XlsxRow(new object?[] { }));
                rows.Add(new XlsxRow(new object?[] { "SẢN PHẨM BÁN CHẠY (xếp theo số lượng bán)" }, XlsxStyle.Bold));
                rows.Add(new XlsxRow(new object?[] { "Hạng", "Sản phẩm", "Số lượng bán", "Doanh thu (VND)" }, XlsxStyle.Bold));

                foreach (var row in _productRows)
                    rows.Add(new XlsxRow(new object?[] { row.Rank, row.ProductName, row.QuantitySold, row.Revenue }));

                WriteSimpleXlsx(dialog.FileName, rows);

                MessageBox.Show("Đã xuất báo cáo Excel (.xlsx) thành công.",
                    "Xuất báo cáo thành công", MessageBoxButton.OK, MessageBoxImage.Information);
            }
            catch (IOException ex)
            {
                MessageBox.Show("Không thể ghi file. Nếu file đang mở trong Excel, hãy đóng lại rồi thử lại.\n\n" + ex.Message,
                    "Lỗi xuất file", MessageBoxButton.OK, MessageBoxImage.Error);
            }
            catch (Exception ex)
            {
                MessageBox.Show("Không thể xuất báo cáo.\n\n" + ex.Message,
                    "Lỗi xuất file", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        // ---------- Ghi file .xlsx tối giản bằng Open XML (không cần NuGet thêm) ----------

        private enum XlsxStyle { Normal = 0, Bold = 1 }

        private sealed record XlsxRow(object?[] Cells, XlsxStyle Style = XlsxStyle.Normal);

        private static void WriteSimpleXlsx(string filePath, IReadOnlyList<XlsxRow> rows)
        {
            using var file = new FileStream(filePath, FileMode.Create, FileAccess.ReadWrite);
            using var archive = new ZipArchive(file, ZipArchiveMode.Create);

            WriteZipText(archive, "[Content_Types].xml",
                "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>" +
                "<Types xmlns=\"http://schemas.openxmlformats.org/package/2006/content-types\">" +
                "<Default Extension=\"rels\" ContentType=\"application/vnd.openxmlformats-package.relationships+xml\"/>" +
                "<Default Extension=\"xml\" ContentType=\"application/xml\"/>" +
                "<Override PartName=\"/xl/workbook.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml\"/>" +
                "<Override PartName=\"/xl/worksheets/sheet1.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml\"/>" +
                "<Override PartName=\"/xl/styles.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.spreadsheetml.styles+xml\"/>" +
                "</Types>");

            WriteZipText(archive, "_rels/.rels",
                "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>" +
                "<Relationships xmlns=\"http://schemas.openxmlformats.org/package/2006/relationships\">" +
                "<Relationship Id=\"rId1\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument\" Target=\"xl/workbook.xml\"/>" +
                "</Relationships>");

            WriteZipText(archive, "xl/workbook.xml",
                "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>" +
                "<workbook xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\" " +
                "xmlns:r=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships\">" +
                "<sheets><sheet name=\"BaoCao\" sheetId=\"1\" r:id=\"rId1\"/></sheets></workbook>");

            WriteZipText(archive, "xl/_rels/workbook.xml.rels",
                "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>" +
                "<Relationships xmlns=\"http://schemas.openxmlformats.org/package/2006/relationships\">" +
                "<Relationship Id=\"rId1\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet\" Target=\"worksheets/sheet1.xml\"/>" +
                "<Relationship Id=\"rId2\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles\" Target=\"styles.xml\"/>" +
                "</Relationships>");

            // cellXfs: 0 = mặc định, 1 = chữ đậm, 2 = số định dạng #,##0 (numFmtId 3)
            WriteZipText(archive, "xl/styles.xml",
                "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>" +
                "<styleSheet xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\">" +
                "<fonts count=\"2\">" +
                "<font><sz val=\"11\"/><name val=\"Calibri\"/></font>" +
                "<font><b/><sz val=\"11\"/><name val=\"Calibri\"/></font>" +
                "</fonts>" +
                "<fills count=\"2\"><fill><patternFill patternType=\"none\"/></fill><fill><patternFill patternType=\"gray125\"/></fill></fills>" +
                "<borders count=\"1\"><border><left/><right/><top/><bottom/><diagonal/></border></borders>" +
                "<cellStyleXfs count=\"1\"><xf numFmtId=\"0\" fontId=\"0\" fillId=\"0\" borderId=\"0\"/></cellStyleXfs>" +
                "<cellXfs count=\"3\">" +
                "<xf numFmtId=\"0\" fontId=\"0\" fillId=\"0\" borderId=\"0\" xfId=\"0\"/>" +
                "<xf numFmtId=\"0\" fontId=\"1\" fillId=\"0\" borderId=\"0\" xfId=\"0\" applyFont=\"1\"/>" +
                "<xf numFmtId=\"3\" fontId=\"0\" fillId=\"0\" borderId=\"0\" xfId=\"0\" applyNumberFormat=\"1\"/>" +
                "</cellXfs>" +
                "</styleSheet>");

            var sheet = new StringBuilder();
            sheet.Append("<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>");
            sheet.Append("<worksheet xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\">");
            sheet.Append("<cols>" +
                         "<col min=\"1\" max=\"1\" width=\"30\" customWidth=\"1\"/>" +
                         "<col min=\"2\" max=\"2\" width=\"34\" customWidth=\"1\"/>" +
                         "<col min=\"3\" max=\"4\" width=\"20\" customWidth=\"1\"/>" +
                         "</cols>");
            sheet.Append("<sheetData>");

            for (int rowIndex = 0; rowIndex < rows.Count; rowIndex++)
            {
                int excelRow = rowIndex + 1;
                XlsxRow xlsxRow = rows[rowIndex];
                sheet.Append($"<row r=\"{excelRow}\">");

                for (int col = 0; col < xlsxRow.Cells.Length; col++)
                {
                    object? value = xlsxRow.Cells[col];
                    if (value == null)
                        continue;

                    string cellRef = GetExcelColumnName(col + 1) + excelRow;
                    int boldStyle = xlsxRow.Style == XlsxStyle.Bold ? 1 : 0;

                    if (value is string text)
                    {
                        string escaped = SecurityElement.Escape(text) ?? string.Empty;
                        sheet.Append($"<c r=\"{cellRef}\" t=\"inlineStr\" s=\"{boldStyle}\"><is><t xml:space=\"preserve\">{escaped}</t></is></c>");
                    }
                    else if (value is decimal money)
                    {
                        sheet.Append($"<c r=\"{cellRef}\" s=\"2\"><v>{money.ToString(CultureInfo.InvariantCulture)}</v></c>");
                    }
                    else if (value is IFormattable number)
                    {
                        sheet.Append($"<c r=\"{cellRef}\"><v>{number.ToString(null, CultureInfo.InvariantCulture)}</v></c>");
                    }
                    else
                    {
                        string escaped = SecurityElement.Escape(value.ToString() ?? string.Empty) ?? string.Empty;
                        sheet.Append($"<c r=\"{cellRef}\" t=\"inlineStr\"><is><t>{escaped}</t></is></c>");
                    }
                }

                sheet.Append("</row>");
            }

            sheet.Append("</sheetData></worksheet>");
            WriteZipText(archive, "xl/worksheets/sheet1.xml", sheet.ToString());
        }

        private static void WriteZipText(ZipArchive archive, string entryName, string content)
        {
            var entry = archive.CreateEntry(entryName, CompressionLevel.Optimal);
            using var stream = entry.Open();
            using var writer = new StreamWriter(stream, new UTF8Encoding(false));
            writer.Write(content);
        }

        private static string GetExcelColumnName(int columnNumber)
        {
            string name = string.Empty;
            while (columnNumber > 0)
            {
                int remainder = (columnNumber - 1) % 26;
                name = (char)('A' + remainder) + name;
                columnNumber = (columnNumber - 1) / 26;
            }
            return name;
        }
